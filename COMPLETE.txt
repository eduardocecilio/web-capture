🚀 COMMANDS RÁPIDOS:

# Setup Local
python -m venv venv
source venv/bin/activate
pip install -r requirements.txt

# Rodar Localmente
python main.py
# Acesse: http://localhost:5000

# Testar API
curl -X POST -H "Content-Type: application/json" \
  -d '{"url":"https://example.com"}' \
  http://localhost:5000/api/convert

# Deploy Vercel
vercel --prod

# Ver Logs
vercel logs --follow
